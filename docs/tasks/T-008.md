## Interfaces (exact)
```ts
export class TimeoutError extends Error {}       // name 'TimeoutError'
export class CircuitOpenError extends Error {}   // name 'CircuitOpenError'; readonly retryAfterMs: number
export class AbortedError extends Error {}       // name 'AbortedError'

export function backoffDelay(attempt: number, opts: { baseMs: number; maxMs: number; random?: () => number }): number;
// full jitter: Math.floor(random() * Math.min(maxMs, baseMs * 2 ** attempt)); attempt starts at 0; random defaults to Math.random

export function sleep(ms: number, signal?: AbortSignal): Promise<void>; // rejects AbortedError on abort, clears its timer

export function retry<T>(
  fn: (attempt: number, signal: AbortSignal) => Promise<T>,
  opts: { retries: number; baseMs: number; maxMs: number; retryOn?: (err: unknown) => boolean;
          signal?: AbortSignal; random?: () => number; onRetry?: (err: unknown, attempt: number, delayMs: number) => void },
): Promise<T>;
// at most retries + 1 calls; rethrows the last error; never retries AbortedError; checks the signal before each attempt and during sleep

export function withTimeout<T>(fn: (signal: AbortSignal) => Promise<T>, ms: number, parent?: AbortSignal): Promise<T>;
// rejects TimeoutError after ms and aborts the inner signal; a parent abort aborts the inner signal and rejects AbortedError; clears its timer when settled

export class CircuitBreaker {
  constructor(opts: { failureThreshold: number; openMs: number; now?: () => number; isFailure?: (err: unknown) => boolean });
  exec<T>(fn: () => Promise<T>): Promise<T>;
  get state(): 'closed' | 'open' | 'half-open';
}
// closed: count consecutive failures (success resets); at the threshold, open.
// open: reject CircuitOpenError immediately (retryAfterMs = remaining open time) until openMs has passed.
// half-open: exactly one probe runs; concurrent calls get CircuitOpenError; probe success closes, failure reopens for openMs.
// isFailure(err) === false lets an error pass through without counting (e.g. 404s).

export class TokenBucket {
  constructor(opts: { capacity: number; refillPerSec: number; now?: () => number });
  tryTake(n?: number): boolean;   // default 1
  msUntil(n?: number): number;    // 0 when available; Infinity when n > capacity
}
```

## Tests to write first (fake timers, injected `now` and `random`)
- **`backoff.spec.ts`:**
  - `is within 0..cap for every attempt` (attempts 0–20, `random = () => 0.999999`);
  - `never exceeds maxMs`;
  - `is 0 with random 0`.
- **`retry.spec.ts`:**
  - `returns the first success without delay`;
  - `stops after retries+1 attempts and rethrows the last error` (`retries: 2`: 3 calls, the last error's message);
  - `does not retry when retryOn returns false`;
  - `aborts promptly when signal aborts during backoff` (rejects `AbortedError`, no further call);
  - `never retries AbortedError`;
  - `calls onRetry with the delay used`.
- **`timeout.spec.ts`:**
  - `resolves when fn finishes in time`;
  - `rejects TimeoutError and aborts the inner signal`;
  - `parent abort rejects AbortedError and aborts the inner signal`;
  - `leaves no pending timers` (`vi.getTimerCount() === 0` after settling).
- **`circuit-breaker.spec.ts`:**
  - `opens after threshold consecutive failures`;
  - `a success resets the failure count`;
  - `rejects fast while open with retryAfterMs`;
  - `half-opens after openMs and closes on probe success`;
  - `reopens on probe failure`;
  - `allows exactly one concurrent probe in half-open`;
  - `ignores errors that isFailure says are not failures`.
- **`token-bucket.spec.ts`:**
  - `starts full`;
  - `refills at rate and caps at capacity`;
  - `msUntil is 0 when tokens available`;
  - `msUntil reports the wait`;
  - `msUntil is Infinity above capacity`.
